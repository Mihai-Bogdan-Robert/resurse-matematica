\topictitle{\tr{Compunerea funcțiilor}{Composition of functions}{Verkettung von Funktionen}}

\tr{Fie}{Let}{Seien}\ $f:A\to B,\ g:B\to C$.\quad
\tr{Funcția}{The function}{Die Funktion}\ $g\circ f:A\to C$,
\[
(g\circ f)(x)=g\bigl(f(x)\bigr)
\]
\tr{se numește „$g$ compus cu $f$''.}{is called ``$g$ composed with $f$''.}{heißt „$g$ verkettet mit $f$``.}
