\topictitle{\tr{Partea întreagă, partea fracționară}{Floor function and fractional part}{Gaußklammer und Nachkommaanteil}}

\subsection{\tr{Partea întreagă}{Integer part (floor)}{Ganzzahlanteil (Gaußklammer)}}
$\ip{x}$ -- \tr{cel mai mare număr \textbf{întreg} mai mic sau egal cu $x$}{the greatest \textbf{integer} less than or equal to $x$}{die größte \textbf{ganze} Zahl, die kleiner oder gleich $x$ ist}
\[
\ip{x}\le x<\ip{x}+1
\]

\subsection{\tr{Partea fracționară}{Fractional part}{Nachkommaanteil}}
\[
\fp{x}=x-\ip{x}\qquad 0\le\fp{x}<1
\]

\subsection{\tr{Proprietăți}{Properties}{Eigenschaften}}
\begin{itemize}
\item $\ip{x+n}=\ip{x}+n\qquad\forall n\in\Z$
\item $\ip{x}+\ip{x+\dfrac12}=\ip{2x}$
\item $\ip{x}+\ip{x+\dfrac1n}+\ip{x+\dfrac2n}+\dots+\ip{x+\dfrac{n-1}{n}}=\ip{nx}\qquad\forall n\in\N^*$\quad(Hermite)
\end{itemize}
