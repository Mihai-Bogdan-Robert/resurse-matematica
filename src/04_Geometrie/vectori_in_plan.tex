\topictitle{\tr{Vectori în plan}{Vectors in the plane}{Vektoren in der Ebene}}

\[
\vec v=x\vec\imath+y\vec\jmath\qquad\mt{sau}{or}{oder}\qquad\vec v(x,y)
\]
\begin{formulae}[1]
\item $\vec u+\vec v=(x_1+x_2)\vec\imath+(y_1+y_2)\vec\jmath$
\item $\vec u-\vec v=(x_1-x_2)\vec\imath+(y_1-y_2)\vec\jmath$
\item $\alpha\cdot\vec u=(\alpha x_1)\vec\imath+(\alpha y_1)\vec\jmath$
\item $\vec u\cdot\vec v=|\vec u|\cdot|\vec v|\cdot\cos(\vec u,\vec v)$\quad \tr{sau}{or}{oder}\quad $\vec u\cdot\vec v=x_1x_2+y_1y_2$
\item $|\vec u|=\sqrt{x_1^2+y_1^2}$
\item $\vect{AB}(x_B-x_A,\ y_B-y_A)$,\qquad $\bigl|\vect{AB}\bigr|=\sqrt{(x_B-x_A)^2+(y_B-y_A)^2}$
\end{formulae}

\begin{enumerate}[label=\arabic*),leftmargin=*,itemsep=6pt]
\item \textbf{\tr{Unghiul dintre doi vectori}{Angle between two vectors}{Winkel zwischen zwei Vektoren}}
  \[\vec v=x\vec\imath+y\vec\jmath,\quad\vec u=x'\vec\imath+y'\vec\jmath\qquad
    \cos(\vec u,\vec v)=\frac{xx'+yy'}{\sqrt{x^2+y^2}\cdot\sqrt{x'^2+y'^2}}\]
  \begin{important}
  $\cos(\vec u,\vec v)<0\imp$ \tr{unghi obtuz}{obtuse angle}{stumpfer Winkel}\qquad
  $\cos(\vec u,\vec v)>0\imp$ \tr{unghi ascuțit}{acute angle}{spitzer Winkel}
  \end{important}
\item \textbf{\tr{Doi vectori coliniari}{Two collinear vectors}{Zwei kollineare Vektoren}}
  \[\vec u\ \mt{și}{and}{und}\ \vec v\ \mt{coliniari}{collinear}{kollinear}\iff\frac x{x'}=\frac y{y'}\ \ (\iff xy'-x'y=0)\]
\item \textbf{\tr{Doi vectori perpendiculari (ortogonali)}{Two perpendicular (orthogonal) vectors}{Zwei senkrechte (orthogonale) Vektoren}}
  \[\vec u\perp\vec v\iff xx'+yy'=0\]
\item \textbf{\tr{Coordonatele punctului care împarte un segment într-un raport dat $k$}{Point dividing a segment in a given ratio $k$}{Punkt, der eine Strecke im gegebenen Verhältnis $k$ teilt}}
  \[\vect{AM}=k\cdot\vect{MB}\ \imp\ \begin{cases}x_M=\dfrac{x_A+k\,x_B}{1+k}\\[8pt] y_M=\dfrac{y_A+k\,y_B}{1+k}\end{cases}\]
\end{enumerate}
