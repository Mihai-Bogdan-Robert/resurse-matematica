\topictitle{\tr{Modulul}{Absolute value}{Betrag}}

\subsection{\tr{Proprietățile modulului}{Properties of the absolute value}{Eigenschaften des Betrags}}
\begin{formulae}
\item $|x|=|-x|\ge 0$
\item $|x|=0\iff x=0$
\item $|x\cdot y|=|x|\cdot|y|$
\item $\left|\dfrac{x}{y}\right|=\dfrac{|x|}{|y|}$\quad $(y\neq 0)$
\end{formulae}
\[
\bigl||x|-|y|\bigr|\;\le\;|x+y|\;\le\;|x|+|y|
\]

\subsection{\tr{Explicitarea modulului}{Piecewise definition}{Fallunterscheidung}}
\[
|x|=\begin{cases} \;\;x, & x\ge 0\\ -x, & x<0\end{cases}
\]
