\topictitle{\tr{Ecuații logaritmice -- formule}{Logarithmic equations -- formulas}{Logarithmusgleichungen -- Formeln}}

\[
\log_a b=x\iff a^x=b\qquad
\bigl(a=\mt{baza}{base}{Basis},\ \ b=\mt{argumentul}{argument}{Numerus}\bigr)
\]

\begin{formulae}
\item $\log_a x+\log_a y=\log_a(xy)$
\item $\log_a x-\log_a y=\log_a\dfrac xy$
\item $\log_a(x^n)=n\log_a x$\quad ($x>0$)
\item $\log_a x=\dfrac{\log_b x}{\log_b a}$
\item $a^{\log_b c}=c^{\log_b a}$
\end{formulae}

\begin{important}
\tr{Un logaritm este definit dacă:}{A logarithm is defined if:}{Ein Logarithmus ist definiert, wenn:}
\quad $\log_a b$:\quad $\begin{cases}a>0\\a\neq1\\b>0\end{cases}$
\end{important}

\subsection{\tr{Tipuri de ecuații}{Types of equations}{Gleichungstypen}}
\begin{enumerate}[label=\Roman*.,leftmargin=*,itemsep=6pt]
\item \tr{De tipul}{Type}{Typ} $\log_{f(x)}g(x)=a$:\quad $\imp g(x)=f(x)^a$
\item \tr{De tipul}{Type}{Typ} $\log_{f(x)}g(x)=\log_{f(x)}h(x)$:\quad $\imp g(x)=h(x)$
\item \tr{Ecuații cu logaritmi în baze diferite: se aduc la aceeași bază.}{Equations with logarithms in different bases: convert to a common base.}{Gleichungen mit Logarithmen verschiedener Basen: auf dieselbe Basis umrechnen.}
\item \tr{Ecuații exponențial-logaritmice.}{Mixed exponential--logarithmic equations.}{Gemischte Exponential-Logarithmus-Gleichungen.}
\item \tr{Ecuații cu logaritmi cu soluție unică:}{Logarithmic equations with a unique solution:}{Logarithmusgleichungen mit genau einer Lösung:}
  \begin{itemize}[label=--,itemsep=1pt]
  \item \tr{primul logaritm se notează cu $a$;}{call the first logarithm $a$;}{den ersten Logarithmus $a$ nennen;}
  \item \tr{al doilea logaritm se notează cu $b$;}{call the second logarithm $b$;}{den zweiten Logarithmus $b$ nennen;}
  \item \tr{se exprimă $x$ în funcție de $a$ și de $b$;}{express $x$ in terms of $a$ and $b$;}{$x$ durch $a$ und $b$ ausdrücken;}
  \item $\dots$
  \end{itemize}
\end{enumerate}
