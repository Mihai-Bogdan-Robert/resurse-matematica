\topictitle{\tr{Ecuații iraționale -- formule}{Irrational equations -- formulas}{Wurzelgleichungen -- Formeln}}

$\sqrt[n]{a}$ \tr{este \textbf{definit} pentru $a\ge0$ (dacă $n$ este \textbf{par}).}{is \textbf{defined} for $a\ge0$ (if $n$ is \textbf{even}).}{ist \textbf{definiert} für $a\ge0$ (wenn $n$ \textbf{gerade} ist).}

\begin{formulae}[3]
\item $\sqrt{a^2}=|a|$
\item $\sqrt[n]{a\cdot b}=\sqrt[n]a\cdot\sqrt[n]b$
\item $\sqrt[n]{\dfrac ab}=\dfrac{\sqrt[n]a}{\sqrt[n]b}$
\item $\sqrt[n]{a^m}=a^{m/n}$
\item $\sqrt[m]{\sqrt[n]a}=\sqrt[m\cdot n]{a}$
\item $\bigl(\sqrt[n]a\bigr)^m=\sqrt[n]{a^m}$
\end{formulae}

\subsection{\tr{Formula radicalilor compuși}{Denesting of radicals}{Verschachtelte Wurzeln auflösen}}
\[
\sqrt{a+\sqrt b}=\sqrt{\frac{a+\sqrt{a^2-b}}{2}}+\sqrt{\frac{a-\sqrt{a^2-b}}{2}}
\]
\[
\sqrt{a-\sqrt b}=\sqrt{\frac{a+\sqrt{a^2-b}}{2}}-\sqrt{\frac{a-\sqrt{a^2-b}}{2}}
\]
